import { createStore } from '@tanstack/react-store'

export interface User {
  id: string
  email: string
  name: string
}

export interface AuthState {
  user: User | null
  token: string | null
  isAuthenticated: boolean
  isHydrated: boolean // 🟩 tracks whether data has been pulled from storage
}

const STORAGE_KEY = 'ts_auth_session'

// Start with a strict, pristine state for the server
export const authStore = createStore<AuthState>({
  user: null,
  token: null,
  isAuthenticated: false,
  isHydrated: false, 
})

// Trigger an explicit client-side hydration immediately on boot
if (typeof window !== 'undefined') {
  try {
    const rawData = localStorage.getItem(STORAGE_KEY)
    if (rawData) {
      const parsed = JSON.parse(rawData)
      if (parsed && typeof parsed === 'object' && 'isAuthenticated' in parsed) {
        // Hydrate the store with stored data
        authStore.setState(() => ({
          ...parsed,
          isHydrated: true,
        }))
      }
    } else {
      // No data found, mark hydration complete anyway
      authStore.setState((prev) => ({ ...prev, isHydrated: true }))
    }
  } catch (err) {
    console.error('Failed to hydrate store:', err)
    authStore.setState((prev) => ({ ...prev, isHydrated: true }))
  }

  // Handle updates cleanly without overwriting during server boot
  authStore.subscribe(() => {
    try {
      const currentState = authStore.state
      // Only write to localStorage if hydration has passed
      if (currentState && currentState.isHydrated) {
        localStorage.setItem(STORAGE_KEY, JSON.stringify({
          user: currentState.user,
          token: currentState.token,
          isAuthenticated: currentState.isAuthenticated
        }))
      }
    } catch (e) {
      console.error('Failed to save auth state:', e)
    }
  })
}

export const authActions = {
  login: (user: User, token: string) => {
    authStore.setState(() => ({
      user,
      token,
      isAuthenticated: true,
      isHydrated: true,
    }))
  },
  logout: () => {
    if (typeof window !== 'undefined') {
      localStorage.removeItem(STORAGE_KEY)
    }
    authStore.setState(() => ({
      user: null,
      token: null,
      isAuthenticated: false,
      isHydrated: true,
    }))
  },
}


/**
 * Resolves once the store has finished loading from localStorage.
 */
export function waitForHydration(): Promise<AuthState> {
  return new Promise((resolve) => {
    // If already hydrated, resolve immediately
    if (authStore.state.isHydrated) {
      return resolve(authStore.state);
    }

    // Otherwise, subscribe and wait for the flag to change
    const unsubscribe = authStore.subscribe(() => {
      if (authStore.state.isHydrated) {
        unsubscribe.unsubscribe();
        resolve(authStore.state);
   
      }
    });
  });
}
