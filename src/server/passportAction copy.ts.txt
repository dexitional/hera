import { createServerFn } from '@tanstack/react-start'
import removeBackground from '@imgly/background-removal-node'
import sharp from 'sharp'
import path from 'path'

const productionAIConfig: any = {
  modelSource: 'local',
  publicPath: typeof window === 'undefined' ? path.resolve('./public/models/background-removal/') : '',
  debug: false
}

export const generatePassportServerFn = createServerFn({ method: 'POST' })
  .handler(async (args: any) => {
    try {
      let imageBase64: any = null

      if (typeof args === 'string') imageBase64 = args
      else if (args?.imageBase64) imageBase64 = args.imageBase64
      else if (args?.data?.imageBase64) imageBase64 = args.data.imageBase64
      else if (args?.data) imageBase64 = args.data

      if (!imageBase64) {
        return { success: false, error: "No image payload found." }
      }

      const base64Data = imageBase64.replace(/^data:image\/\w+;base64,/, '')
      const initialBuffer = Buffer.from(base64Data, 'base64')

      // 1. Clear out orientation metadata and convert to valid PNG bytes
      const orientedInputBuffer:any = await sharp(initialBuffer)
        .rotate() 
        .png() 
        .toBuffer()

      const inputBlob = new Blob([orientedInputBuffer], { type: 'image/png' })

      // 2. Run the AI background removal tool
      let transparentImageBuffer: Buffer
      try {
        const transparentBlob = await removeBackground(inputBlob)
        const bufferArray = await transparentBlob.arrayBuffer()
        transparentImageBuffer = Buffer.from(bufferArray)
      } catch (aiError) {
        console.error("AI Background Removal failed, using fallback trim:", aiError)
        transparentImageBuffer = orientedInputBuffer
      }

      /**
       * 🛠️ CRITICAL FIX: AUTO-TRIM DEAD SPACE
       * We use .trim() to automatically crop away any surrounding black or white 
       * empty spaces. This isolates the man's head and shoulders into a tight bounding box,
       * ensuring our math starts from his actual profile dimensions.
       */
      const trimmedBuffer = await sharp(transparentImageBuffer)
        .trim({ threshold: 10 }) // Cuts out solid surrounding empty backgrounds safely
        .toBuffer()

      // 3. Gather clean dimensions of the isolated subject profile
      const metadata = await sharp(trimmedBuffer).metadata()
      const width = metadata.width || 400
      const height = metadata.height || 400

      /**
       * 📏 PASSPORT BIOMETRIC SCALE MATH
       * We create standard clearance padding around his isolated body profile.
       */
      const targetSize = Math.max(width, height)
    //   const topPadding = Math.floor(targetSize * 0.14)    // 14% headspace above hair
      const topPadding = Math.floor(targetSize * 0.05)    // 5% headspace above hair
      const bottomPadding = Math.floor(targetSize * 0.0005) 
      const sidePadding = Math.floor((targetSize - width) / 40)

      // 4. Construct final official 600x600px white background passport
      const processedBuffer = await sharp(trimmedBuffer)
        .extend({
          top: topPadding,
          bottom: bottomPadding,
          left: sidePadding > 0 ? sidePadding : 10,
          right: sidePadding > 0 ? sidePadding : 10,
          background: { r: 255, g: 255, b: 255, alpha: 1 } // Solid Plain White Background
        })
        .flatten({ background: { r: 255, g: 255, b: 255 } }) // Ensures no transparent gaps remain
        .resize(600, 600, {
          fit: 'cover',
          position: 'top' // Keeps framing anchored gracefully on head/shoulders
        })
        .jpeg({ quality: 95 })
        .toBuffer()

      const outputBase64 = `data:image/jpeg;base64,${processedBuffer.toString('base64')}`
      return { success: true, passportImageUrl: outputBase64 }

    } catch (error: any) {
      console.error('Passport Processing Server Error:', error)
      return { success: false, error: error.message || 'Failed to process photo.' }
    }
  })
