import DashboardHeader from '#/components/DashboardHeader';
import { createFileRoute, Link } from '@tanstack/react-router'
import { Boxes, Plus, Search } from "lucide-react";
import { useState } from "react";

// Sample mock data simulating backend responses
const INITIAL_ORGANIZATIONS = [
  {
    id: "knvmlleenlmznzfsltwy",
    name: "Capevars.com",
    plan: "Free Plan",
    projectsCount: 2,
    href: "/dashboard/org/knvmlleenlmznzfsltwy",
  },
];

export const Route = createFileRoute('/admin/organizations/index copy')({
  component: OrganizationsDashboard,
})

export default function OrganizationsDashboard() {
  const [searchQuery, setSearchQuery] = useState("");
  const [organizations] = useState(INITIAL_ORGANIZATIONS);

  // Filter organizations down via reactive inputs
  const filteredOrganizations = organizations.filter((org) =>
    org.name.toLowerCase().includes(searchQuery.toLowerCase()),
  );

  return (
    <div className="min-h-screen bg-[#18181b] text-white antialiased font-sans flex flex-col items-center">
     
    <main className="w-full flex flex-col items-center justify-center">
      {/* Header Panel */}
      <div className="@container max-w-[1200px] px-4 @lg:px-6 @xl:px-10 w-full mx-auto pt-12">
        <div className="space-y-4">
          <div className="flex items-center justify-between gap-4">
            <div className="space-y-4">
              <div className="flex items-center gap-4">
                <div className="space-y-1">
                  <h1 className="text-2xl font-bold tracking-tight text-white">
                    Your Organizations
                  </h1>
                </div>
              </div>
            </div>
            <div className="flex items-center gap-2"></div>
          </div>
        </div>
      </div>

      {/* Main Dashboard Section */}
      <div className="mx-auto w-full @container max-w-[1200px] px-4 @lg:px-6 @xl:px-10">
        <div className="first:pt-12 py-6 w-full flex flex-col gap-y-4">
          {/* Action Ribbon: Search & Add button */}
          <div className="flex items-center justify-between gap-x-2 md:gap-x-3">
            {/* Search Input Container Wrapper */}
            <div
              data-slot="input-group"
              role="group"
              className="group/input-group relative items-center outline-hidden transition-[color,box-shadow] flex rounded-md border border-zinc-700 bg-zinc-900 text-sm w-full md:w-64"
            >
              <input
                data-slot="input-group-control"
                type="text"
                placeholder="Search for an organization"
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                className="placeholder:text-zinc-500 text-xs px-2.5 py-1 h-[26px] block rounded-none border border-transparent bg-transparent shadow-none focus:outline-none focus:border-transparent w-full flex-1 text-white"
              />
              <div
                role="group"
                data-slot="input-group-addon"
                data-align="inline-start"
                className="text-zinc-400 flex h-auto cursor-text select-none items-center justify-center gap-2 text-sm order-first pl-2"
              >
                <Search className="w-3.5 h-3.5" />
              </div>
            </div>

            {/* Creation Trigger Action */}
            <Link
              data-size="tiny"
              className="relative justify-center cursor-pointer inline-flex items-center space-x-2 text-center font-regular ease-out duration-200 rounded-md border bg-purple-600 hover:bg-purple-500 text-white border-purple-500/75 text-xs px-2.5 py-1 h-[26px] w-min"
              to={`/admin/organizations/new`}
            >
              <div className="inline-flex items-center justify-center shrink-0 text-white">
                <Plus className="w-3.5 h-3.5" />
              </div>
              <span className="truncate">New organization</span>
            </Link>
          </div>

          {/* Cards Display System */}
          <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
            {filteredOrganizations.length > 0 ? (
              filteredOrganizations.map((org) => (
                <a key={org.id} href={org.href}>
                  <div className="overflow-hidden rounded-lg text-white shadow-xs grow bg-zinc-900/50 p-3 transition-colors hover:bg-zinc-800/80 border border-zinc-800 hover:border-zinc-700 cursor-pointer flex items-center min-h-[70px] max-h-min">
                    <div className="relative flex items-start gap-3 w-full">
                      {/* Graphics Node */}
                      <div className="rounded-full bg-zinc-800 border border-zinc-700 w-8 h-8 flex items-center justify-center shrink-0">
                        <Boxes className="w-4 h-4 text-zinc-300" />
                      </div>

                      {/* Content Section */}
                      <div className="grow flex flex-col gap-0 min-w-0">
                        <div className="flex items-center gap-x-2">
                          <h3
                            title={org.name}
                            className="text-sm font-medium text-white mb-0 truncate max-w-full"
                          >
                            {org.name}
                          </h3>
                        </div>
                        <div className="flex items-center justify-between text-xs text-zinc-400 font-sans">
                          <div className="flex items-center gap-x-1">
                            <span>{org.plan}</span>
                            <span className="text-zinc-600">·</span>
                            <span>{org.projectsCount} projects</span>
                          </div>
                          <div className="flex items-center gap-x-2"></div>
                        </div>
                      </div>
                    </div>
                  </div>
                </a>
              ))
            ) : (
              <div className="col-span-full text-center py-8 text-zinc-500 text-sm">
                No organizations found matching your search.
              </div>
            )}
          </div>
        </div>
      </div>
      </main>
    </div>
  );
}
