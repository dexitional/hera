import { createFileRoute, Link } from "@tanstack/react-router";
import React, { useState, useEffect } from "react";
import { 
  UserPlus, Upload, Search, Filter, Edit2, Trash2, 
  CheckCircle, XCircle, ChevronDown, FileSpreadsheet, X, 
  Loader2, Smartphone, ChevronLeft, ChevronRight 
} from "lucide-react";
import { useMutation, useQueryClient, useSuspenseQuery } from "@tanstack/react-query";
import { deleteVoterFn, getVotersByElectionFn, inviteVoterFn, inviteVotersFn, uploadVotersFn } from "#/server/tenant-elections";
import * as XLSX from 'xlsx';
import { generateSixDigitCode } from "#/lib/utils";

// 1. Explicit Parameter Schema Type
type VotersSearchSchema = {
  page: number;
  search: string;
  status: "ALL" | "VOTED" | "PENDING";
};

// 2. Query Key Definition Factory
const electionsQueryOptions = (params: {
  electionId: number;
  page: number;
  search: string;
  statusFilter: "ALL" | "VOTED" | "PENDING";
}) => ({
  queryKey: ['voters-admin', params.electionId, params.page, params.search, params.statusFilter],
  queryFn: () => getVotersByElectionFn({
    electionId: params.electionId,
    page: params.page,
    pageSize: 25,
    searchQuery: params.search,
    statusFilter: params.statusFilter,
  }),
});

// 3. Setup Configuration with Correct Escaped Route Path Symbol
export const Route = createFileRoute("/admin/elections/$electionId/voters")({
  validateSearch: (search: Record<string, unknown>): VotersSearchSchema => {
    return {
      page: Number(search?.page) || 1,
      search: (search?.search as string) || "",
      status: (search?.status as "ALL" | "VOTED" | "PENDING") || "ALL",
    };
  },
  loader: async ({ context, params, search }) => {
    const electionId = Number(params.electionId);
    const page = search?.page ?? 1;
    const searchStr = search?.search ?? "";
    const statusFilter = search?.statusFilter ?? "ALL";
    
    await context.queryClient.ensureQueryData(
      electionsQueryOptions({ electionId, page, search: searchStr, statusFilter })
    );
  },
  pendingComponent: () => (
    <div className="flex justify-center items-center h-64 w-full">
      <Loader2 className="animate-spin text-purple-500 w-8 h-8" />
    </div>
  ),
  component: VotersDirectory,
});

function VotersDirectory() {
  const queryClient = useQueryClient();
  
  // Clean hook invocation safely bound to the route identity string context
  const navigate = Route.useNavigate();
  const { electionId } = Route.useParams();
  const numElectionId = Number(electionId);
  console.log("numElectionId", numElectionId);
  
  const searchParams = Route.useSearch() || {};
  const page = Number(searchParams.page) || 1;
  const search = searchParams.search || "";
  const status = searchParams.status || "ALL";

  const [localSearch, setLocalSearch] = useState(search);
  const [isExcelModalOpen, setIsExcelModalOpen] = useState(false);
  const [excelFile, setExcelFile] = useState<File | null>(null);

  // Sync state if parameters change via external history transitions
  useEffect(() => {
    setLocalSearch(search);
  }, [search]);

  // Read data from React Query with complete reactive parameter dependencies
  const { data: apiResponse }: any = useSuspenseQuery(
    electionsQueryOptions({ electionId: numElectionId, page, search, statusFilter: status })
  );

  const serverResult = apiResponse?.success ? apiResponse : { data: [], meta: { totalItems: 0, totalPages: 1 } };
  const votersList = serverResult.data || [];
  const { totalItems, totalPages } = serverResult.meta || { totalItems: 0, totalPages: 1 };

  // Search input change debouncer function
  useEffect(() => {
    if (localSearch === search) return;

    const delayDebounceFn = setTimeout(() => {
      navigate({
        search: (prev) => ({
          ...prev,
          page: 1, // Drop back to page 1 on fresh search filtering strings
          search: localSearch,
        }),
      });
    }, 400);

    return () => clearTimeout(delayDebounceFn);
  }, [localSearch, search, navigate]);

  // ================= MUTATIONS OPERATIONS =================
  const deleteMutation = useMutation({
    mutationFn: deleteVoterFn,
    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['voters-admin'] }); },
    onError: (error: any) => console.error(error.message)
  });

  const inviteMutation = useMutation({
    mutationFn: inviteVoterFn,
    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['voters-admin'] }); },
    onError: (error: any) => console.error(error.message)
  });

  const invitesMutation = useMutation({
    mutationFn: inviteVotersFn,
    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['voters-admin'] }); },
    onError: (error: any) => console.error(error.message)
  });

  const importExcelMutation = useMutation({
    mutationFn: uploadVotersFn,
    onSuccess: () => { queryClient.invalidateQueries({ queryKey: ['voters-admin'] }); },
    onError: (error: any) => console.error(error.message)
  });

  // ================= HANDLERS INTERACTION INTERFACES =================
  const handleDeleteVoter = (id: any) => {
    if (confirm("Are you sure you want to remove this voter?")) {
       deleteMutation.mutate({ data: id });
    }
  };

  const handleInviteVoter = (id: any) => {
    if (confirm("Invite this voter?")) {
      inviteMutation.mutate({ data: id });
    }
  };

  const handleInviteVoters = () => {
    if (confirm("Send bulk invitations?")) {
      invitesMutation.mutate({ data: numElectionId } as any);
    }
  };

  const handleFilterStatusChange = (newStatus: "ALL" | "VOTED" | "PENDING") => {
    navigate({
      search: (prev) => ({
        ...prev,
        page: 1, // Drop back safely to page 1 on active classification switches
        status: newStatus,
      }),
    });
  };

  const handlePageChange = (targetPage: number) => {
    if (targetPage >= 1 && targetPage <= totalPages) {
      navigate({
        search: (prev) => ({
          ...prev,
          page: targetPage,
        }),
      });
    }
  };

  const handleExcelUploadSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!excelFile) return;

    const reader = new FileReader();
    reader.readAsArrayBuffer(excelFile);
    reader.onload = (e) => {
      try {
        const data = e.target?.result;
        const workbook = XLSX.read(data, { type: 'array' });
        const firstSheetName = workbook.SheetNames[0];
        const worksheet = workbook.Sheets[firstSheetName];
        const rawJson = XLSX.utils.sheet_to_json<any>(worksheet);
        
        const sanitizedJson = rawJson.map((row) => {
          let rawPhone = String(row?.phone || '').trim();
          rawPhone = rawPhone.replace(/[\s\-\(\)]/g, '');
          const cleanPhone = rawPhone.startsWith('233') ? "0"+(rawPhone.slice(3)) : rawPhone;

          return {
            username: row.username,
            name: row.name?.trim(),
            phoneNumber: cleanPhone || '0000000000',
            electionId: numElectionId,
            inviteToken: generateSixDigitCode(),
            email: row?.email ? row?.email?.trim() : `${row?.username?.replaceAll(" ","")}@vote.local`
          };
        });

        importExcelMutation.mutate({ data: sanitizedJson } as any);
      } catch (error) {
        console.error('Error reading Excel file:', error);
      } finally {
        setIsExcelModalOpen(false);
        setExcelFile(null);
      }
    };
  };


  return (
    <>
      <div className="max-w-7xl mx-auto space-y-6 text-zinc-100">
        
        {/* ================= HEADER RIBBON CONTROLS ================= */}
        <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 bg-[#0a192a]/50 p-6 rounded-xl border border-zinc-800">
          <div>
            <h1 className="text-xl font-bold text-white tracking-tight">Voters Manager</h1>
            <p className="text-xs text-zinc-400 mt-1">
              Manage eligibility list frameworks, trace authentication access keys, and send invitation codes to voters.
            </p>
          </div>
          
          <div className="flex items-center gap-3 shrink-0">
            <button
              onClick={handleInviteVoters}
              disabled={invitesMutation.isPending}
              className="flex items-center gap-2 bg-zinc-900 hover:bg-zinc-800 border border-zinc-800 text-xs px-3.5 py-2 rounded-lg font-medium text-zinc-300 transition-all disabled:opacity-50"
            >
              {invitesMutation.isPending ? <Loader2 className="w-3 h-3 animate-spin text-purple-400" /> : <Smartphone className="w-3 h-3 text-purple-400" />}
              <span>Send Invites</span>
            </button>

            <button
              onClick={() => setIsExcelModalOpen(true)}
              className="flex items-center gap-2 bg-zinc-900 hover:bg-zinc-800 border border-zinc-800 text-xs px-3.5 py-2 rounded-lg font-medium text-zinc-300 transition-all"
            >
              <FileSpreadsheet className="w-4 h-4 text-emerald-500" />
              <span>Import Excel</span>
            </button>

            <Link
              to="/admin/voters/new"
              className="flex items-center gap-2 bg-purple-600 hover:bg-purple-500 text-white text-xs px-3.5 py-2 rounded-lg font-medium shadow-md transition-all"
            >
              <UserPlus className="w-4 h-4" />
              <span>Create Voter</span>
            </Link>
          </div>
        </div>

        {/* ================= FILTERS & QUICK CONTROLS AREA ================= */}
        <div className="flex flex-col sm:flex-row gap-4 justify-between items-center bg-zinc-900/50 p-4 rounded-xl border border-zinc-800/80">
          
          {/* Left Column: Instant Controlled Text Input Buffer */}
          <div className="relative w-full sm:max-w-xs">
            <Search className="absolute left-3 top-2.5 h-4 w-4 text-zinc-500" />
            <input
              type="text"
              placeholder="Search voter registry..."
              value={localSearch}
              onChange={(e) => setLocalSearch(e.target.value)}
              className="w-full bg-zinc-950 border border-zinc-800 rounded-lg pl-9 pr-10 py-2 text-xs text-zinc-100 placeholder-zinc-500 focus:outline-none focus:border-purple-500 transition-colors"
            />

            
            {/* Dynamic clear button context indicator */}
            {localSearch && (
              <button 
                type="button"
                onClick={() => setLocalSearch("")} 
                className="absolute right-3 top-2.5 text-zinc-500 hover:text-zinc-300 transition-colors"
              >
                <X className="w-3.5 h-3.5" />
              </button>
            )}
          </div>

          {/* Right Column: Attendance Selector Framework Controls */}
          <div className="flex items-center gap-2 w-full sm:w-auto justify-end">
            <Filter className="w-3.5 h-3.5 text-zinc-400" />
            <div className="relative">
              <select
                value={status}
                onChange={(e) => handleFilterStatusChange(e.target.value as "ALL" | "VOTED" | "PENDING")}
                className="appearance-none bg-zinc-950 border border-zinc-800 rounded-lg pl-3 pr-10 py-2 text-xs text-zinc-300 focus:outline-none focus:border-purple-500 cursor-pointer transition-colors"
              >
                <option value="ALL">All Attendance Status</option>
                <option value="VOTED">Voted Status Completed</option>
                <option value="PENDING">Pending Ballots Outstanding</option>
              </select>
              
              {/* Absolute positioning visual arrow anchor decoration overlay */}
              <div className="pointer-events-none absolute inset-y-0 right-0 flex items-center px-2 text-zinc-500 border-l border-zinc-800/60 my-1">
                <ChevronDown className="h-3 w-3" />
              </div>
            </div>
          </div>

        </div>

        {/* ================= MAIN DATA MATRIX COMPONENT TABLE ================= */}
        <div className="bg-zinc-900/30 border border-zinc-800 rounded-xl overflow-hidden shadow-xl">
          <div className="overflow-x-auto">
            <table className="w-full text-left border-collapse">
              <thead>
                <tr className="bg-zinc-900/80 border-b border-zinc-800 text-zinc-400 font-medium text-[11px] uppercase tracking-wider">
                  <th className="py-3 px-4">Voter Identity</th>
                  <th className="py-3 px-4">Access Username</th>
                  <th className="py-3 px-4">Contact Info</th>
                  <th className="py-3 px-4">Status Map</th>
                  <th className="py-3 px-4 text-center">Security Hash Key</th>
                  <th className="py-3 px-4 text-right">Actions Panel</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-zinc-800/60 text-xs">
                {votersList.length === 0 ? (
                  <tr>
                    <td colSpan={6} className="py-12 text-center text-zinc-500">
                      No matching records found inside this election scope parameters.
                    </td>
                  </tr>
                ) : (
                  votersList.map((row: any) => {
                    const voter = row.voters;
                    return (
                      <tr key={voter.id} className="hover:bg-zinc-800/20 transition-colors">
                        <td className="py-3.5 px-4 font-medium text-white">{voter.name}</td>
                        <td className="py-3.5 px-4 text-zinc-400">{voter.username}</td>
                        <td className="py-3.5 px-4 space-y-0.5">
                          <div className="text-zinc-300">{voter.phoneNumber}</div>
                          <div className="text-[10px] text-zinc-500">{voter.email}</div>
                        </td>
                        <td className="py-3.5 px-4">
                          {voter.hasVoted ? (
                            <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400 font-medium text-[10px] border border-emerald-500/20">
                              <CheckCircle className="w-2.5 h-2.5" /> Voted
                            </span>
                          ) : (
                            <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full bg-amber-500/10 text-amber-400 font-medium text-[10px] border border-amber-500/20">
                              <XCircle className="w-2.5 h-2.5" /> Pending
                            </span>
                          )}
                        </td>
                        <td className="py-3.5 px-4 text-center font-mono text-xs text-purple-400">
                          {voter.inviteToken}
                        </td>
                        <td className="py-3.5 px-4 text-right space-x-2">
                          <button
                            onClick={() => handleInviteVoter(voter.id)}
                            disabled={inviteMutation.isPending}
                            className="p-1.5 rounded bg-zinc-900 border border-zinc-800 text-zinc-400 hover:text-purple-400 hover:border-purple-500/30 transition-colors disabled:opacity-50"
                            title="Send individual code invite"
                          >
                            <Smartphone className="w-3.5 h-3.5" />
                          </button>
                          <button
                            onClick={() => handleDeleteVoter(voter.id)}
                            disabled={deleteMutation.isPending}
                            className="p-1.5 rounded bg-zinc-900 border border-zinc-800 text-zinc-400 hover:text-rose-400 hover:border-rose-500/30 transition-colors disabled:opacity-50"
                            title="Remove voter entry record"
                          >
                            <Trash2 className="w-3.5 h-3.5" />
                          </button>
                        </td>
                      </tr>
                    );
                  })
                )}
              </tbody>
            </table>
          </div>

          {/* ================= PAGINATION NAVIGATION CONTROL FRAMEWORK ================= */}
          <div className="flex flex-col sm:flex-row items-center justify-between gap-4 bg-zinc-950 p-4 border-t border-zinc-800 text-xs text-zinc-400">
            <div>
              Showing <span className="font-semibold text-white">{totalItems === 0 ? 0 : (Number(page) - 1) * 25 + 1}</span> to{" "}
              <span className="font-semibold text-white">{Math.min(Number(page) * 25, totalItems)}</span> of{" "}
              <span className="font-semibold text-white">{totalItems}</span> voters
            </div>

            <div className="flex items-center gap-1.5">
              {/* Previous Page Control Button */}
              <button
                onClick={() => handlePageChange(Number(page) - 1)}
                disabled={Number(page) === 1}
                className="p-2 rounded-lg bg-zinc-900 border border-zinc-800 hover:bg-zinc-800 text-zinc-300 disabled:opacity-30 disabled:hover:bg-zinc-900 disabled:cursor-not-allowed transition-all"
              >
                <ChevronLeft className="w-3.5 h-3.5" />
              </button>

              {/* Smart Page Truncation Engine */}
              {(() => {
                const pageRange = 1; 
                const pages: (number | string)[] = [];
                const currentPageNum = Number(page);

                for (let i = 1; i <= totalPages; i++) {
                  if (
                    i === 1 || 
                    i === totalPages || 
                    (i >= currentPageNum - pageRange && i <= currentPageNum + pageRange)
                  ) {
                    pages.push(i);
                  } else if (pages[pages.length - 1] !== "...") {
                    pages.push("...");
                  }
                }

                return pages.map((pageNum, index) => {
                  if (pageNum === "...") {
                    return (
                      <span key={`ellipsis-${index}`} className="px-2 text-zinc-600 select-none font-medium">
                        ...
                      </span>
                    );
                  }

                  return (
                    <button
                      key={`page-${pageNum}`}
                      onClick={() => handlePageChange(pageNum as number)}
                      className={`px-3 py-1.5 rounded-lg font-medium text-xs transition-all border ${
                        currentPageNum === pageNum
                          ? "bg-purple-600 border-purple-500 text-white shadow-md"
                          : "bg-zinc-900 border-zinc-800 text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200"
                      }`}
                    >
                      {pageNum}
                    </button>
                  );
                });
              })()}

              {/* Next Page Control Button */}
              {/* <button
                onClick={() => handlePageChange(currentPageNum + 1)}
                disabled={currentPageNum === totalPages || totalItems === 0}
                className="p-2 rounded-lg bg-zinc-900 border border-zinc-800 hover:bg-zinc-800 text-zinc-300 disabled:opacity-30 disabled:hover:bg-zinc-900 disabled:cursor-not-allowed transition-all"
              >
                <ChevronRight className="w-3.5 h-3.5" />
              </button> */}
            </div>
          </div>
          
        </div>
      </div>
    
    </>)
    
  }
