import React, { useState } from 'react';
import { useMutation, useQuery } from '@tanstack/react-query';
import Papa from 'papaparse';
import { dispatchUpgradedInvites, getElectionTurnoutStats } from '../../server/admin';

export function CompleteVoterManagerPanel({ electionId }: { electionId: number }) {
  const [isParsing, setIsParsing] = useState(false);

  // Poll real-time audit headcount metrics from the database every 5 seconds
  const { data: auditData } = useQuery({
    queryKey: ['electionAudit', electionId],
    queryFn: () => getElectionTurnoutStats(electionId),
    refetchInterval: 5000,
  });

  const inviteMutation = useMutation({
    mutationFn: dispatchUpgradedInvites,
    onSuccess: (res: any) => alert(res.message),
    onError: (err: any) => alert(`Import failed: ${err.message}`)
  });

  // Client-Side CSV Row Parser Engine
  const handleCsvFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    setIsParsing(true);

    Papa.parse(file, {
      header: true, // Auto-maps columns by matching header row labels (e.g., "name", "username", "phone")
      skipEmptyLines: true,
      complete: (results: any) => {
        setIsParsing(false);
        
        // Structure raw sheet rows into type-safe database objects
        const formattedVoters = results.data.map((row: any) => ({
          name: row.name || row['Full Name'] || '',
          username: row.username || row['Username'] || '',
          phoneNumber: row.phone || row['Phone Number'] || row.phoneNumber || '',
        })).filter((v: any) => v.name && v.username && v.phoneNumber);

        if (formattedVoters.length === 0) {
          alert('Could not find columns matching: name, username, phone.');
          return;
        }

        // Fire array payload directly down to your TanStack Server Function
        inviteMutation.mutate({ electionId, targetVoters: formattedVoters } as any);
      },
      error: (err: any) => {
        setIsParsing(false);
        alert(`CSV parse structural error: ${err.message}`);
      }
    });
  };

  return (
    <div className="space-y-8 max-w-4xl mx-auto p-6">
      
      {/* Live Election Turnout Statistics View Box */}
      <div className="bg-white p-6 rounded-2xl border border-gray-200 grid grid-cols-3 gap-4 shadow-sm text-center">
        <div>
          <p className="text-xs font-semibold text-gray-400 uppercase tracking-wider">Registered</p>
          <p className="text-2xl font-black text-gray-800">{auditData?.registered || 0}</p>
        </div>
        <div>
          <p className="text-xs font-semibold text-gray-400 uppercase tracking-wider">Ballots Cast</p>
          <p className="text-2xl font-black text-blue-600">{auditData?.voted || 0}</p>
        </div>
        <div>
          <p className="text-xs font-semibold text-gray-400 uppercase tracking-wider">Turnout Rate</p>
          <p className="text-2xl font-black text-green-600">{auditData?.turnoutPercentage || 0}%</p>
        </div>
      </div>

      {/* Excel/CSV Direct Spreadsheet Loader Drag Widget */}
      <div className="bg-white p-8 rounded-2xl border border-gray-200 shadow-sm">
        <h3 className="text-lg font-bold text-gray-900 mb-1">Bulk Upload Voter Sheet</h3>
        <p className="text-xs text-gray-500 mb-6">
          Upload a standard <code className="bg-gray-100 px-1 py-0.5 rounded text-red-600 font-mono">.csv</code> file containing columns labeled exactly as: <strong className="font-semibold text-gray-700">name</strong>, <strong className="font-semibold text-gray-700">username</strong>, and <strong className="font-semibold text-gray-700">phone</strong>.
        </p>

        <div className="border-2 border-dashed border-gray-200 rounded-xl p-8 text-center hover:border-blue-400 transition relative bg-gray-50">
          <input 
            type="file" 
            accept=".csv" 
            onChange={handleCsvFileUpload}
            disabled={isParsing || inviteMutation.isPending}
            className="absolute inset-0 w-full h-full opacity-0 cursor-pointer"
          />
          <div className="space-y-2">
            <p className="text-sm font-semibold text-gray-700">
              {isParsing ? 'Processing sheet records...' : inviteMutation.isPending ? 'Sending SMS alerts...' : 'Drop your CSV spreadsheet file here'}
            </p>
            <p className="text-xs text-gray-400">or click to browse local computer directory files</p>
          </div>
        </div>
      </div>

    </div>
  );
}
