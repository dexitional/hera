import React, { useState } from 'react';
import { authClient } from '../../lib/auth-client';

export function AdminAuthPortal() {
  const [isSignUp, setIsSignUp] = useState(false);
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  const handleAuthenticationSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setIsLoading(true);

    try {
      if (isSignUp) {
        // Create a new administrator profile in your database via Better Auth
        await authClient.signUp.email({
          email,
          password,
          name,
          callbackURL: '/admin/dashboard'
        });
        alert('Account registered successfully!');
      } else {
        // Authenticate credentials against stored cryptographic password hashes
        await authClient.signIn.email({
          email,
          password,
          callbackURL: '/admin/dashboard'
        });
      }
    } catch (err: any) {
      alert(`Authentication failed: ${err.message || 'Check your credentials.'}`);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="max-w-md mx-auto my-20 p-8 bg-white border rounded-2xl shadow-xl">
      <h2 className="text-2xl font-black text-gray-900 mb-2">
        {isSignUp ? 'Create Admin Account' : 'Admin Portal Sign-In'}
      </h2>
      <p className="text-xs text-gray-400 mb-6">Manage events, organizational elections, and live voter registries</p>

      <form onSubmit={handleAuthenticationSubmit} className="space-y-4">
        {isSignUp && (
          <div>
            <label className="block text-xs font-semibold text-gray-600 uppercase mb-1">Full Name</label>
            <input 
              type="text" required value={name} onChange={(e) => setName(e.target.value)}
              className="w-full bg-gray-50 border rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
              placeholder="e.g., Chief Election Officer"
            />
          </div>
        )}
        <div>
          <label className="block text-xs font-semibold text-gray-600 uppercase mb-1">Email Address</label>
          <input 
            type="email" required value={email} onChange={(e) => setEmail(e.target.value)}
            className="w-full bg-gray-50 border rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
            placeholder="admin@organization.org"
          />
        </div>
        <div>
          <label className="block text-xs font-semibold text-gray-600 uppercase mb-1">Security Password</label>
          <input 
            type="password" required value={password} onChange={(e) => setPassword(e.target.value)}
            className="w-full bg-gray-50 border rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
            placeholder="••••••••"
          />
        </div>

        <button 
          type="submit" disabled={isLoading}
          className="w-full bg-blue-600 text-white font-bold py-3 rounded-xl text-sm transition hover:bg-blue-700 disabled:bg-gray-200"
        >
          {isLoading ? 'Verifying Profile Gateway...' : isSignUp ? 'Register Account' : 'Secure Login'}
        </button>
      </form>

      <div className="mt-6 text-center border-t pt-4">
        <button 
          onClick={() => setIsSignUp(!isSignUp)}
          className="text-xs font-semibold text-blue-600 hover:underline bg-transparent border-0 cursor-pointer"
        >
          {isSignUp ? 'Already registered? Sign in here' : 'Need an administrative account? Sign up here'}
        </button>
      </div>
    </div>
  );
}
