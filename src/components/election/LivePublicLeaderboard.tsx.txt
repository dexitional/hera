import React from 'react';
import { useQuery } from '@tanstack/react-query';
import { getCachedElectionResults } from '../../server/results';

export function LivePublicLeaderboard({ electionId }: { electionId: number }) {
  const { data: report, isLoading } = useQuery({
    queryKey: ['publicResults', electionId],
    queryFn: () => getCachedElectionResults(electionId),
    refetchInterval: 5000, // Client component polls the server function safely every 5 seconds
    staleTime: 4500, // Tells TanStack Query not to issue redundant re-fetches within this tight browser lifecycle window
  });

  if (isLoading) return <p className="text-center py-6 text-gray-400 text-sm">Loading live election statistics...</p>;
  if (!report) return null;

  return (
    <div className="max-w-2xl mx-auto p-6 bg-white border border-gray-150 rounded-2xl shadow-sm">
      <div className="mb-6 border-b border-gray-100 pb-3">
        <h2 className="text-xl font-extrabold text-gray-900">{report.electionTitle}</h2>
        <p className="text-xs font-semibold text-green-600 flex items-center gap-1.5 mt-0.5">
          <span className="w-2 h-2 rounded-full bg-green-500 animate-pulse" /> Live updates active
        </p>
      </div>

      {report.positions.map((pos: any, idx: number) => (
        <div key={idx} className="mb-8">
          <h3 className="text-sm font-bold text-gray-500 uppercase tracking-wider mb-3">Office: {pos.title}</h3>
          <div className="space-y-4">
            {pos.candidates.map((cand: any) => {
              const sharePercent = pos.totalPositionVotes > 0 ? ((cand.votes / pos.totalPositionVotes) * 100) : 0;
              return (
                <div key={cand.id} className="relative">
                  <div className="flex justify-between text-sm font-semibold text-gray-700 mb-1">
                    <span>{cand.name}</span>
                    <span className="font-mono text-gray-900">{cand.votes} votes ({sharePercent.toFixed(1)}%)</span>
                  </div>
                  <div className="w-full bg-gray-100 h-2.5 rounded-full overflow-hidden">
                    <div className="bg-blue-600 h-2.5 rounded-full transition-all duration-700 ease-out" style={{ width: `${sharePercent}%` }} />
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      ))}
    </div>
  );
}
