import React from 'react';
import { useQuery } from '@tanstack/react-query';
import { getFullElectionResults } from '../../server/results';

export function ElectionResultsDashboard({ electionId }: { electionId: number }) {
  const { data: report, isLoading } = useQuery({
    queryKey: ['electionResults', electionId],
    queryFn: () => getFullElectionResults(electionId)
  });

  // 1. Export Data Directly to a CSV Spreadsheet
  const exportToSpreadsheet = () => {
    if (!report) return;
    
    let csvContent = "data:text/csv;charset=utf-8,Position,Candidate Name,Votes Received,Percentage Share\n";
    
    report.positions.forEach((pos: any) => {
      pos.candidates.forEach((cand: any) => {
        const share = pos.totalPositionVotes > 0 ? ((cand.votes / pos.totalPositionVotes) * 100).toFixed(1) : 0;
        csvContent += `"${pos.title}","${cand.name}",${cand.votes},${share}%\n`;
      });
    });

    const encodedUri = encodeURI(csvContent);
    const link = document.createElement("a");
    link.setAttribute("href", encodedUri);
    link.setAttribute("download", `${report.electionTitle.replace(/\s+/g, '_')}_Official_Results.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  if (isLoading) return <p className="text-center py-12 text-sm text-gray-500">Compiling official tallies...</p>;
  if (!report) return null;

  return (
    <div className="max-w-4xl mx-auto p-6 space-y-8">
      {/* Control Actions Section (Hidden on Print Outlay) */}
      <div className="print:hidden flex justify-end gap-4 bg-gray-50 p-4 rounded-xl border">
        <button 
          onClick={exportToSpreadsheet}
          className="bg-white border border-gray-300 text-gray-700 font-bold px-4 py-2 rounded-lg text-sm hover:bg-gray-100 transition"
        >
          📂 Export Excel CSV
        </button>
        <button 
          onClick={() => window.print()}
          className="bg-blue-600 text-white font-bold px-4 py-2 rounded-lg text-sm hover:bg-blue-700 transition"
        >
          🖨️ Print Official Report
        </button>
      </div>

      {/* Printable Report Canvas Area */}
      <div className="bg-white p-8 print:p-0 rounded-2xl border border-gray-150 print:border-0 shadow-sm print:shadow-none">
        
        {/* Print Report Header Block */}
        <div className="border-b-2 border-gray-900 pb-4 mb-8 text-center sm:text-left">
          <h1 className="text-3xl font-black text-gray-900 uppercase tracking-tight">{report.electionTitle}</h1>
          <p className="text-sm font-semibold text-gray-500 mt-1">Official Certified Election Report • Generated on {new Date().toLocaleDateString()}</p>
        </div>

        {/* Dynamic Category Layout Mapping Loops */}
        {report.positions.map((pos: any, idx: number) => (
          <div key={idx} className="mb-12 page-break-inside-avoid">
            <div className="bg-gray-900 text-white px-4 py-2 rounded-t-lg print:rounded-none flex justify-between font-bold text-sm uppercase">
              <span>Office: {pos.title}</span>
              <span>Total Votes: {pos.totalPositionVotes}</span>
            </div>

            <table className="w-full border-collapse border border-gray-200 mt-0 text-left">
              <thead>
                <tr className="bg-gray-50 border-b border-gray-200 text-xs font-bold text-gray-500 uppercase">
                  <th className="p-3 w-20">Photo</th>
                  <th className="p-3">Candidate</th>
                  <th className="p-3 w-32 text-right">Votes</th>
                  <th className="p-3 w-40 text-right">Share Bar</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-gray-150 text-sm">
                {pos.candidates.map((cand: any) => {
                  const sharePercent = pos.totalPositionVotes > 0 ? ((cand.votes / pos.totalPositionVotes) * 100) : 0;
                  return (
                    <tr key={cand.id} className="hover:bg-gray-50/50 print:hover:bg-transparent">
                      <td className="p-3">
                        {cand.photo ? (
                          <img 
                            src={cand.photo} 
                            alt={cand.name} 
                            className="w-12 h-12 object-cover rounded-md border border-gray-200" 
                          />
                        ) : (
                          <div className="w-12 h-12 bg-gray-100 rounded-md flex items-center justify-center text-xs text-gray-400">No Image</div>
                        )}
                      </td>
                      <td className="p-3 font-semibold text-gray-800">{cand.name}</td>
                      <td className="p-3 text-right font-bold text-gray-900">{cand.votes}</td>
                      <td className="p-3 text-right">
                        <div className="flex items-center justify-end gap-3">
                          <span className="font-mono text-xs font-bold text-gray-600">{sharePercent.toFixed(1)}%</span>
                          <div className="w-24 bg-gray-100 h-2 rounded-full overflow-hidden print:hidden">
                            <div className="bg-gray-800 h-2 rounded-full" style={{ width: `${sharePercent}%` }} />
                          </div>
                        </div>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        ))}

        {/* Certificate Sign-off Area (Visible only when physically printed) */}
        <div className="hidden print:block mt-24 border-t border-dashed border-gray-400 pt-6">
          <div className="flex justify-between text-xs text-gray-500 font-mono">
            <div>
              <p className="border-b border-black w-48 h-8 mb-1"></p>
              <p>Certified By: Election Administrator Signature</p>
            </div>
            <div className="text-right">
              <p>Verification Stamp Code: SECURE_BALLOT_{electionId}</p>
            </div>
          </div>
        </div>

      </div>
    </div>
  );
}
