import React, { useState } from 'react';
import { useQuery, useMutation } from '@tanstack/react-query';
import { submitWebBallot } from '../../server/admin';

interface Candidate { id: number; name: string; imageUrl?: string; }
interface Position { id: number; title: string; candidates: Candidate[]; }

export function SecureWebBallotPage({ electionTitle, positionsList, token, electionId }: {
  electionTitle: string;
  positionsList: Position[];
  token: string;
  electionId: number;
}) {
  // Keep track of user selections: { [positionId]: candidateId }
  const [ ballot, setBallot ] = useState<Record<number, number>>({});

  const mutation = useMutation({
    mutationFn: () => submitWebBallot({
      token,
      electionId,
      selections: Object.entries(ballot).map(([posId, candId]) => ({
        positionId: Number(posId),
        candidateId: candId,
      })),
    }),
    onSuccess: (res:any) => alert(res.message),
    onError: (err: any) => alert(err.message),
  });

  const handleSelect = (positionId: number, candidateId: number) => {
    setBallot((prev) => ({ ...prev, [positionId]: candidateId }));
  };

  const isBallotComplete = positionsList.every((pos) => ballot[pos.id] !== undefined);

  return (
    <div className="max-w-2xl mx-auto my-12 p-8 bg-white border border-gray-200 rounded-2xl shadow-xl">
      <div className="border-b border-gray-100 pb-4 mb-6">
        <span className="text-xs font-bold text-red-600 bg-red-50 px-2.5 py-1 rounded-full uppercase tracking-wider">
          Secure Secret Ballot
        </span>
        <h1 className="text-2xl font-extrabold text-gray-900 mt-2">{electionTitle}</h1>
      </div>

      {positionsList.map((pos) => (
        <div key={pos.id} className="mb-8 p-6 bg-gray-50 rounded-xl border border-gray-100">
          <h3 className="text-lg font-bold text-gray-800 mb-4">{pos.title}</h3>
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            {pos.candidates.map((cand) => {
              const isSelected = ballot[pos.id] === cand.id;
              return (
                <button
                  key={cand.id}
                  onClick={() => handleSelect(pos.id, cand.id)}
                  className={`flex items-center gap-4 p-4 rounded-xl border text-left transition-all ${
                    isSelected 
                      ? 'bg-blue-50 border-blue-500 ring-2 ring-blue-200' 
                      : 'bg-white border-gray-200 hover:border-gray-300'
                  }`}
                >
                  <div className={`w-5 h-5 rounded-full border flex items-center justify-center ${isSelected ? 'border-blue-600 bg-blue-600' : 'border-gray-300'}`}>
                    {isSelected && <div className="w-2 h-2 rounded-full bg-white" />}
                  </div>
                  <span className="font-semibold text-gray-800">{cand.name}</span>
                </button>
              );
            })}
          </div>
        </div>
      ))}

      <button
        onClick={() => mutation.mutate()}
        disabled={!isBallotComplete || mutation.isPending}
        className="w-full mt-4 bg-gray-900 text-white font-bold py-4 rounded-xl shadow-md transition-all hover:bg-black disabled:bg-gray-200 disabled:text-gray-400 disabled:cursor-not-allowed"
      >
        {mutation.isPending ? 'Encrypting & Submitting...' : 'Submit Final Ballot'}
      </button>
    </div>
  );
}
