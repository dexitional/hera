import React, { useState } from 'react';
import { authClient } from '../../lib/auth-client';

export function AdminTwoFactorSetup() {
  const [qrCodeUrl, setQrCodeUrl] = useState<string | null>(null);
  const [totpSecret, setTotpSecret] = useState('');
  const [verificationCode, setVerificationCode] = useState('');
  const [is2faEnabled, setIs2faEnabled] = useState(false);

  // 1. Initialize the 2FA Activation Process
  const handleEnable2FA = async () => {
    try {
      // Requests Better Auth to generate a unique cryptographically safe TOTP secret
      const res = await authClient.twoFactor.enable({
        password: "ADMIN_CONFIRMATION_PASSWORD" // Ask the admin to re-enter their password for safety
      });
      
      if (res?.data) {
        setQrCodeUrl(res.data.qrCode); // Base64 image string to render on the client side
        setTotpSecret(res.data.secret);
      }
    } catch (err: any) {
      alert(`Initialization error: ${err.message}`);
    }
  };

  // 2. Verify and Lock the Authenticator Setup Hook
  const handleVerifyAndActivate = async (e: React.FormEvent) => {
    e.preventDefault();
    try {
      await authClient.twoFactor.verify({
        code: verificationCode
      });
      setIs2faEnabled(true);
      alert('🔒 Two-Factor Authentication has been successfully activated on your profile!');
    } catch (err: any) {
      alert(`Invalid verification token code: ${err.message}`);
    }
  };

  return (
    <div className="max-w-md mx-auto p-6 bg-white border border-gray-200 rounded-2xl shadow-sm">
      <h3 className="text-lg font-black text-gray-900 mb-1">Account Security Level</h3>
      <p className="text-xs text-gray-500 mb-6">Add an extra layer of protection to your elections workspace dashboard.</p>

      {is2faEnabled ? (
        <div className="bg-green-50 border border-green-200 p-4 rounded-xl text-center">
          <p className="text-sm font-bold text-green-800">✅ 2FA Protection Active</p>
        </div>
      ) : qrCodeUrl ? (
        <form onSubmit={handleVerifyAndActivate} className="space-y-4 text-center">
          <p className="text-xs text-gray-600 font-semibold mb-2">Scan this QR code with Google Authenticator or 1Password:</p>
          <img src={qrCodeUrl} alt="TOTP QR Code" className="mx-auto w-48 h-48 border p-2 bg-white rounded-lg" />
          <p className="text-xs text-gray-400 font-mono select-all">Secret: {totpSecret}</p>
          
          <div className="text-left">
            <label className="block text-xs font-bold text-gray-600 uppercase mb-1">Enter 6-Digit Verification Code</label>
            <input 
              type="text" required maxLength={6} value={verificationCode} 
              onChange={(e) => setVerificationCode(e.target.value)}
              className="w-full text-center bg-gray-50 border tracking-widest text-lg font-bold rounded-lg px-3 py-2 focus:outline-none focus:ring-2 focus:ring-blue-500"
              placeholder="000000"
            />
          </div>
          <button type="submit" className="w-full bg-blue-600 text-white font-bold py-2.5 rounded-xl text-sm transition hover:bg-blue-700">
            Confirm & Activate 2FA
          </button>
        </form>
      ) : (
        <button 
          onClick={handleEnable2FA}
          className="w-full bg-gray-900 text-white font-bold py-2.5 rounded-xl text-sm transition hover:bg-black"
        >
          🔐 Setup Two-Factor Authenticator App
        </button>
      )}
    </div>
  );
}
