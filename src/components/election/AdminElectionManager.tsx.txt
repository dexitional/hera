import { useMutation } from "@tanstack/react-query";
import React, { useState } from "react";
import { createCandidate, dispatchVoterInvites } from "../../server/admin";
import { getCandidatePhotoUploadUrl } from "../../server/storage";

export function AdminElectionManager({
  electionId,
  positionId,
}: {
  electionId: number;
  positionId: number;
}) {
  // Local Element Component States
  const [candidateName, setCandidateName] = useState("");
  const [selectedFile, setSelectedFile] = useState<File | null>(null);
  const [voterPhonesRaw, setVoterPhonesRaw] = useState("");
  const [isUploading, setIsUploading] = useState(false);

  // Mutation to create the final candidate row
  const candidateMutation = useMutation({
    mutationFn: createCandidate,
    onSuccess: () => {
      alert("Candidate added cleanly!");
      setCandidateName("");
      setSelectedFile(null);
    },
  } as any);

  // Mutation to dispatch SMS invitations
  const inviteMutation = useMutation({
    mutationFn: dispatchVoterInvites,
    onSuccess: (res: any) => alert(res.message),
  } as any);

  // Execute Photo Upload to Cloudflare R2 / S3
  const handleAddCandidate = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!candidateName) return;

    let finalPublicImageUrl = "";

    if (selectedFile) {
      setIsUploading(true);
      try {
        // 1. Get Secure Presigned URL from TanStack Server Function
        const { uploadUrl, publicUrl } = await getCandidatePhotoUploadUrl({
          fileName: selectedFile.name,
          fileType: selectedFile.type,
        });
        // 2. Put binary data straight onto the bucket storage layer
        await fetch(uploadUrl, {
          method: "PUT",
          body: selectedFile,
          headers: { "Content-Type": selectedFile.type },
        });

        finalPublicImageUrl = publicUrl;
      } catch (err) {
        console.error("File direct streaming upload failure:", err);
        alert("Image sync issue encountered.");
        setIsUploading(false);
        return;
      }
      setIsUploading(false);
    }
    // 3. Commit row values down to PostgreSQL using Drizzle
    candidateMutation.mutate({
      positionId,
      name: candidateName,
      imageUrl: finalPublicImageUrl,
    } as any);
  };

  const handleSendInvites = (e: React.FormEvent) => {
    e.preventDefault();
    // Parse commas or newline values from text area inputs
    const phoneNumbers = voterPhonesRaw
      .split(/[\n,]+/)
      .map((p) => p.trim())
      .filter((p) => p.length > 0);

    if (phoneNumbers.length === 0) return;

    inviteMutation.mutate({ electionId, phoneNumbers } as any);
    setVoterPhonesRaw("");
  };

  return (
    <div className="grid grid-cols-1 md:grid-cols-2 gap-8 max-w-5xl mx-auto p-6">
      {/* Box A: Add Candidate Form panel with File Upload hook */}
      <div className="bg-white p-6 rounded-2xl border border-gray-200 shadow-sm">
        <h2 className="text-xl font-bold text-gray-900 mb-4">
          Add Position Candidate
        </h2>
        <form onSubmit={handleAddCandidate} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-gray-600 uppercase mb-1">
              Candidate Name
            </label>
            <input
              type="text"
              value={candidateName}
              onChange={(e) => setCandidateName(e.target.value)}
              className="w-full bg-gray-50 border rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
              placeholder="e.g. John Mahama Kwesi"
            />
          </div>
          <div>
            <label className="block text-xs font-semibold text-gray-600 uppercase mb-1">
              Profile Photo (S3/R2)
            </label>
            <input
              type="file"
              accept="image/*"
              onChange={(e) => setSelectedFile(e.target.files?.[0] || null)}
              className="w-full text-sm text-gray-500 file:mr-4 file:py-2 file:px-4 file:rounded-full file:border-0 file:text-xs file:font-semibold file:bg-blue-50 file:text-blue-700 hover:file:bg-blue-100"
            />
          </div>
          <button
            type="submit"
            disabled={isUploading || candidateMutation.isPending}
            className="w-full bg-blue-600 text-white font-bold py-2.5 rounded-lg text-sm transition hover:bg-blue-700 disabled:bg-gray-200"
          >
            {isUploading ? "Streaming Photo to R2..." : "Save Candidate"}
          </button>
        </form>
      </div>

      {/* Box B: Import Voters Panel and SMS Blast Engine */}
      <div className="bg-white p-6 rounded-2xl border border-gray-200 shadow-sm">
        <h2 className="text-xl font-bold text-gray-900 mb-4">
          Enfranchise List of Voters
        </h2>
        <form onSubmit={handleSendInvites} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-gray-600 uppercase mb-1">
              Voter Phone Registry
            </label>
            <textarea
              rows={4}
              value={voterPhonesRaw}
              onChange={(e) => setVoterPhonesRaw(e.target.value)}
              className="w-full bg-gray-50 border rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500 font-mono"
              placeholder="+233541112222&#10;+233203334444"
            />
            <p className="text-xs text-gray-400 mt-1">
              Separate telephone inputs using commas or plain newlines.
            </p>
          </div>
          <button
            type="submit"
            disabled={inviteMutation.isPending}
            className="w-full bg-gray-900 text-white font-bold py-2.5 rounded-lg text-sm transition hover:bg-black disabled:bg-gray-200"
          >
            {inviteMutation.isPending
              ? "Queuing SMS Actions..."
              : "Dispatch Unique Ballot Invites"}
          </button>
        </form>
      </div>
    </div>
  );
}
