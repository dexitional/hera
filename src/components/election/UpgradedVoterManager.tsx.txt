import React, { useState } from 'react';
import { useMutation, useQuery } from '@tanstack/react-query';
import { dispatchUpgradedInvites, getElectionTurnoutStats } from '../../server/admin';

export function UpgradedVoterManager({ electionId }: { electionId: number }) {
  const [voterInputRaw, setVoterInputRaw] = useState('');

  // Live Auditing Metrics hook using TanStack Query polling configurations
  const { data: auditData } = useQuery({
    queryKey: ['electionAudit', electionId],
    queryFn: () => getElectionTurnoutStats(electionId),
    refetchInterval: 5000, // Pull current headcount updates every 5 seconds
  });

  const inviteMutation = useMutation({
    mutationFn: dispatchUpgradedInvites,
    onSuccess: (res: any) => {
      alert(res.message);
      setVoterInputRaw('');
    },
  });

  const handleImportVoters = (e: React.FormEvent) => {
    e.preventDefault();

    // Map rows matching format structural layout parameters: Name, Username, Phone
    const lines = voterInputRaw.split('\n');
    const targetVoters = lines
      .map((line) => {
        const parts = line.split(',');
        if (parts.length < 3) return null;
        return {
          name: parts[0].trim(),
          username: parts[1].trim(),
          phoneNumber: parts[2].trim(),
        };
      })
      .filter((v): v is { name: string; username: string; phoneNumber: string } => v !== null);

    if (targetVoters.length === 0) {
      alert('Please use the correct input layout formatting.');
      return;
    }

    inviteMutation.mutate({ electionId, targetVoters } as any);
  };

  return (
    <div className="space-y-8 max-w-4xl mx-auto p-6">
      {/* Turnout Auditing Widget Banner */}
      <div className="bg-white p-6 rounded-2xl border border-gray-200 grid grid-cols-3 gap-4 shadow-sm text-center">
        <div>
          <p className="text-xs font-semibold text-gray-400 uppercase">Registered Voters</p>
          <p className="text-2xl font-bold text-gray-800">{auditData?.registered || 0}</p>
        </div>
        <div>
          <p className="text-xs font-semibold text-gray-400 uppercase">Ballots Received</p>
          <p className="text-2xl font-bold text-blue-600">{auditData?.voted || 0}</p>
        </div>
        <div>
          <p className="text-xs font-semibold text-gray-400 uppercase">Live Turnout</p>
          <p className="text-2xl font-bold text-green-600">{auditData?.turnoutPercentage || 0}%</p>
        </div>
      </div>

      {/* Upgraded Multi-Attribute Import Entry Card */}
      <div className="bg-white p-8 rounded-2xl border border-gray-200 shadow-sm">
        <h2 className="text-lg font-bold text-gray-900 mb-2">Import Voter Accounts</h2>
        <p className="text-xs text-gray-500 mb-4">
          Add one voter per line using the following layout format: <code className="bg-gray-100 px-1 py-0.5 rounded text-red-600 font-mono">Full Name, Username, Phone Number</code>
        </p>

        <form onSubmit={handleImportVoters} className="space-y-4">
          <textarea
            rows={6}
            value={voterInputRaw}
            onChange={(e) => setVoterInputRaw(e.target.value)}
            className="w-full bg-gray-50 border border-gray-200 rounded-xl px-4 py-3 text-sm font-mono focus:outline-none focus:ring-2 focus:ring-blue-500"
            placeholder="John Doe, j_doe, +233540001111&#10;Jane Mensah, jane_m, +233200002222"
          />

          <button
            type="submit"
            disabled={inviteMutation.isPending}
            className="w-full bg-gray-900 text-white font-bold py-3 rounded-xl text-sm transition hover:bg-black disabled:bg-gray-200"
          >
            {inviteMutation.isPending ? 'Validating Profiles & Dispatching Links...' : 'Register & Send SMS Invites'}
          </button>
        </form>
      </div>
    </div>
  );
}
