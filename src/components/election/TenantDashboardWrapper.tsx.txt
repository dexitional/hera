import React, { useState, useEffect } from 'react';
import { useQuery } from '@tanstack/react-query';
import { getAdminOrganizations, getTenantElections } from '../../server/tenant';

export function TenantDashboardWrapper() {
  const [activeOrgId, setActiveOrgId] = useState<number | null>(null);

  // 1. Load all organizations this administrator has administrative rights over
  const { data: orgs, isLoading: loadingOrgs } = useQuery({
    queryKey: ['adminOrgs'],
    queryFn: () => getAdminOrganizations(),
  });

  // Automatically default to the first organization if none is selected
  useEffect(() => {
    if (orgs && orgs.length > 0 && !activeOrgId) {
      setActiveOrgId(orgs[0].id);
    }
  }, [orgs, activeOrgId]);

  // 2. Dynamically fetch workspace elections strictly when activeOrgId changes
  const { data: electionsList, isLoading: loadingElections } = useQuery({
    queryKey: ['tenantElections', activeOrgId],
    queryFn: () => getTenantElections(activeOrgId!),
    enabled: activeOrgId !== null,
  });

  if (loadingOrgs) return <p className="text-center py-12 text-sm text-gray-500">Initializing workspaces...</p>;

  return (
    <div className="min-h-screen bg-gray-50 flex flex-col">
      {/* Top Multi-Tenant Context Command Banner */}
      <header className="bg-white border-b border-gray-200 px-6 py-4 flex items-center justify-between shadow-sm">
        <div className="flex items-center gap-4">
          <span className="text-sm font-bold text-gray-400 uppercase tracking-wider">Active Workspace:</span>
          <select
            value={activeOrgId || ''}
            onChange={(e) => setActiveOrgId(Number(e.target.value))}
            className="bg-gray-100 border border-gray-300 text-gray-800 font-bold px-3 py-1.5 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
          >
            {orgs?.map((org: any) => (
              <option key={org.id} value={org.id}>
                🏢 {org.name}
              </option>
            ))}
          </select>
        </div>
        <div className="text-xs font-semibold text-gray-500">Tenant Context Isolation Isolation Enabled</div>
      </header>

      {/* Primary Workspace Dashboard Body */}
      <main className="flex-1 p-8 max-w-5xl w-full mx-auto">
        <h2 className="text-xl font-black text-gray-900 mb-4">Workspace Elections Registry</h2>
        
        {loadingElections ? (
          <p className="text-sm text-gray-400">Loading scoped election records...</p>
        ) : electionsList && electionsList.length > 0 ? (
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            {electionsList.map((el:any) => (
              <div key={el.id} className="bg-white p-6 rounded-xl border border-gray-150 shadow-sm flex flex-col justify-between">
                <div>
                  <h4 className="font-bold text-gray-800 text-lg">{el.title}</h4>
                  <p className="text-xs text-gray-400 mt-1">Ends: {new Date(el.endAt).toLocaleString()}</p>
                </div>
                <div className="mt-4 pt-4 border-t flex justify-between items-center">
                  <span className={`text-xs px-2.5 py-0.5 rounded-full font-bold ${el.isActive ? 'bg-green-50 text-green-700' : 'bg-gray-100 text-gray-600'}`}>
                    {el.isActive ? 'Active' : 'Closed'}
                  </span>
                  <button className="text-xs bg-gray-900 text-white font-semibold px-3 py-1.5 rounded-md hover:bg-black transition">
                    Manage Ballot
                  </button>
                </div>
              </div>
            ))}
          </div>
        ) : (
          <div className="text-center py-12 bg-white rounded-xl border border-dashed border-gray-200 text-gray-400 text-sm">
            No elections found for this workspace. Create one to get started.
          </div>
        )}
      </main>
    </div>
  );
}
