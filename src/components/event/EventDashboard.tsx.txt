import React, { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { getCategoryLeaderboard } from '../../server/functions';

interface Category {
  id: number;
  name: string;
}

interface DashboardProps {
  eventId: number;
  categoriesList: Category[]; // Passed down from TanStack Route loader
}

export function EventDashboard({ eventId, categoriesList }: DashboardProps) {
  // 1. Manage state for the active selected category
  const [selectedCategoryId, setSelectedCategoryId] = useState<number>(
    categoriesList[0]?.id || 0
  );

  // 2. Fetch data automatically using TanStack Query
  const { data: contestants, isLoading } = useQuery({
    queryKey: ['leaderboard', eventId, selectedCategoryId],
    queryFn: () => getCategoryLeaderboard({ eventId, categoryId: selectedCategoryId }),
    enabled: selectedCategoryId > 0,
    refetchInterval: 3000, // Poll and refresh data every 3 seconds
  });

  return (
    <div className="max-w-3xl mx-auto p-6 bg-gray-50 rounded-xl shadow-sm">
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center mb-6 gap-4">
        <div>
          <h1 className="text-2xl font-bold text-gray-900">Live Standings</h1>
          <p className="text-sm text-gray-500">Real-time vote tallies from Web & USSD</p>
        </div>

        {/* Category Dropdown Selector */}
        <div className="w-full sm:w-64">
          <label className="block text-xs font-semibold text-gray-600 uppercase mb-1">
            Filter Category
          </label>
          <select
            value={selectedCategoryId}
            onChange={(e) => setSelectedCategoryId(Number(e.target.value))}
            className="w-full bg-white border border-gray-300 rounded-lg px-3 py-2 text-sm text-gray-800 focus:outline-none focus:ring-2 focus:ring-blue-500"
          >
            {categoriesList.map((cat) => (
              <option key={cat.id} value={cat.id}>
                {cat.name}
              </option>
            ))}
          </select>
        </div>
      </div>

      {/* Render Leaderboard Standings */}
      {isLoading ? (
        <div className="text-center py-12 text-gray-500 text-sm">Synchronizing data charts...</div>
      ) : (
        <div className="space-y-4 bg-white p-6 rounded-lg border border-gray-150">
          {contestants && contestants.length > 0 ? (
            contestants.map((player: any) => (
              <div key={player.id} className="relative">
                <div className="flex justify-between text-sm font-medium text-gray-700 mb-1">
                  <span>{player.name}</span>
                  <span className="font-bold text-blue-600">{player.totalVotes || 0} votes</span>
                </div>
                {/* Visual Progress Bar */}
                <div className="w-full bg-gray-100 rounded-full h-3 overflow-hidden">
                  <div
                    className="bg-blue-600 h-3 rounded-full transition-all duration-500 ease-out"
                    style={{
                      width: `${Math.min(player.totalVotes || 0, 100)}%`,
                    }}
                  />
                </div>
              </div>
            ))
          ) : (
            <p className="text-center text-sm text-gray-400 py-6">No contestants registered yet.</p>
          )}
        </div>
      )}
    </div>
  );
}
